💡 使用示例
───────────────────────────────────────────────────────────────────────

# 打开背包
UIManager.open_ui("inventory")

# 切换装备面板
UIManager.toggle_ui("equipment")

# 显示通知
UIManager.hud.show_notification("获得新物品！", 2.0)

# 显示升级效果
UIManager.hud.show_level_up(5)

# 监听 UI 事件
UIManager.ui_opened.connect(func(ui_name):
    print("打开了: ", ui_name)
)
